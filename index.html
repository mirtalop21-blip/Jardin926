/** * TITANIUM 360 - SISTEMA INTEGRAL DE GESTIÓN (AUDITORÍA J926) * CÓDIGO MAESTRO COMPLETO - GITHUB REPO */ let DATA = { mat: [], esp: [], bajas: [], asistencia: [] }; // --- 1. MOTOR DE PERSISTENCIA Y CARGA --- function sync() { localStorage.setItem('T360_DB_FINAL', JSON.stringify(DATA)); renderSupervision(); renderEspera(); renderAsistencia(); } function importData(input) { const reader = new FileReader(); reader.onload = (e) => { try { const raw = JSON.parse(e.target.result); DATA.mat = Array.isArray(raw) ? raw : (raw.mat || []); sync(); alert("BASE DE DATOS TITANIUM 360 CARGADA CON ÉXITO."); } catch (err) { alert("Error de formato JSON."); } }; reader.readAsText(input.files[0]); } // --- 2. LEGAJO 360 & RADAR --- function smartRadar() { const val = document.getElementById('radar').value.toUpperCase(); const res = DATA.mat.find(a => a.alu_nombre.includes(val) || a.alu_dni === val); if(res) { Object.keys(res).forEach(k => { const el = document.getElementById(k); if(el) el.value = res[k]; }); show(1); } } // --- 3. SUPERVISIÓN ÉLITE (Cálculos Matemáticos) --- function renderSupervision() { const total = DATA.mat.length; if (total === 0) return; const stats = { ninos: DATA.mat.filter(a => a.sexo === 'M').length, ninas: DATA.mat.filter(a => a.sexo === 'F').length, arg: DATA.mat.filter(a => a.alu_nac === 'ARG').length, par: DATA.mat.filter(a => a.alu_nac === 'PAR').length, bol: DATA.mat.filter(a => a.alu_nac === 'BOL').length, per: DATA.mat.filter(a => a.alu_nac === 'PER').length, auh: DATA.mat.filter(a => a.alu_auh === 'SI').length, cud: DATA.mat.filter(a => a.cud === 'SI').length }; // Renderizado en DOM (ID's necesarios) document.getElementById('header_total').innerText = total; console.log("Supervisión Élite Actualizada:", stats); } // --- 4. LISTA DE ESPERA PRO --- function registrarEspera(alu) { DATA.esp.push({...alu, fecha: new Date().toLocaleDateString()}); sync(); } // --- 5. BITÁCORA DE BAJAS --- function registrarBaja(dni, motivo) { const idx = DATA.mat.findIndex(x => x.alu_dni === dni); if(idx > -1) { DATA.bajas.push({...DATA.mat[idx], motivo, fecha: new Date().toLocaleDateString()}); DATA.mat.splice(idx, 1); sync(); } } // --- 6. VOZ BIÓNICA --- const recognition = new (window.SpeechRecognition || window.webkitSpeechRecognition)(); recognition.lang = 'es-ES'; function iniciarCargaVoz() { recognition.start(); recognition.onresult = (e) => { const transcript = e.results[0][0].transcript.toUpperCase(); console.log("Comando voz:", transcript); // Implementación de lógica de comandos: ej "Buscar DNI..." }; } // --- 7. NAVEGACIÓN --- function show(tab) { document.querySelectorAll('main > section').forEach((s, i) => s.classList.toggle('hidden', i !== (tab - 1))); document.querySelectorAll('.btn-tab').forEach((b, i) => b.classList.toggle('active-tab', i === (tab - 1))); } window.onload = () => { const saved = localStorage.getItem('T360_DB_FINAL'); if(saved) { DATA = JSON.parse(saved); renderSupervision(); } };
